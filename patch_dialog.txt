package com.example.ui.components

import androidx.compose.foundation.background
import androidx.compose.foundation.border
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.LazyRow
import androidx.compose.foundation.lazy.items
import androidx.compose.foundation.shape.CircleShape
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.foundation.text.KeyboardOptions
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.filled.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.draw.clip
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.testTag
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.text.input.KeyboardType
import androidx.compose.ui.text.style.TextAlign
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import androidx.compose.ui.window.Dialog
import androidx.compose.ui.window.DialogProperties
import com.example.model.BellSlot
import com.example.model.ClassSlot
import com.example.model.WeekParity
import com.example.model.subjectColorPalette
import com.example.ui.theme.*
import java.time.DayOfWeek
import java.time.LocalTime
import java.time.format.DateTimeFormatter
import java.time.format.TextStyle
import java.util.Locale
import java.util.UUID

data class DayClassSetting(
    val day: DayOfWeek,
    var startTime: String,
    var endTime: String
)

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun AddEditClassDialog(
    initialSlot: ClassSlot? = null,
    defaultDay: DayOfWeek = DayOfWeek.MONDAY,
    bellSlots: List<BellSlot>,
    onDismiss: () -> Unit,
    onSave: (List<ClassSlot>) -> Unit,
    onDelete: (String) -> Unit = {}
) {
    val isEditMode = initialSlot != null
    var subjectTitle by remember { mutableStateOf(initialSlot?.subjectTitle ?: "") }
    var professor by remember { mutableStateOf(initialSlot?.professor ?: "") }
    var classroom by remember { mutableStateOf(initialSlot?.classroom ?: "") }
    var classType by remember { mutableStateOf(initialSlot?.classType ?: "Лекция") }
    
    val timeFormatter = DateTimeFormatter.ofPattern("HH:mm")
    
    var daySettings by remember {
        mutableStateOf(
            if (isEditMode) {
                listOf(DayClassSetting(
                    day = initialSlot!!.dayOfWeek,
                    startTime = initialSlot.startTime.format(timeFormatter),
                    endTime = initialSlot.endTime.format(timeFormatter)
                ))
            } else {
                listOf(DayClassSetting(
                    day = defaultDay,
                    startTime = "10:00",
                    endTime = "11:30"
                ))
            }
        )
    }
    
    var weekParity by remember { mutableStateOf(initialSlot?.weekParity ?: WeekParity.ALL) }
    var selectedColorHex by remember { mutableStateOf(initialSlot?.colorHex ?: subjectColorPalette.first().first) }
    var errorMessage by remember { mutableStateOf<String?>(null) }

    Dialog(
        onDismissRequest = onDismiss,
        properties = DialogProperties(usePlatformDefaultWidth = false, dismissOnClickOutside = false)
    ) {
        Surface(
            shape = RoundedCornerShape(24.dp),
            color = BentoSurface,
            modifier = Modifier
                .fillMaxWidth(0.95f)
                .fillMaxHeight(0.95f)
                .padding(vertical = 16.dp)
        ) {
            Column(modifier = Modifier.fillMaxSize()) {
                // Header
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .background(BentoSurfaceVariant.copy(alpha = 0.5f))
                        .padding(horizontal = 16.dp, vertical = 12.dp),
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.SpaceBetween
                ) {
                    IconButton(onClick = onDismiss) {
                        Icon(Icons.Default.ArrowBack, contentDescription = "Назад", tint = BentoOnSurface)
                    }
                    Text(
                        text = if (isEditMode) "Редактировать занятие" else "Добавить занятие",
                        style = MaterialTheme.typography.titleLarge.copy(fontWeight = FontWeight.Bold),
                        color = BentoOnSurface
                    )
                    IconButton(
                        onClick = {
                            if (subjectTitle.isBlank()) {
                                errorMessage = "Введите название предмета"
                                return@IconButton
                            }
                            
                            val newSlots = mutableListOf<ClassSlot>()
                            
                            for (setting in daySettings) {
                                val st = try {
                                    LocalTime.parse(setting.startTime, timeFormatter)
                                } catch (e: Exception) {
                                    errorMessage = "Неверный формат времени (HH:mm) для ${setting.day.getDisplayName(TextStyle.SHORT, Locale("ru"))}"
                                    return@IconButton
                                }
                                val et = try {
                                    LocalTime.parse(setting.endTime, timeFormatter)
                                } catch (e: Exception) {
                                    errorMessage = "Неверный формат времени (HH:mm) для ${setting.day.getDisplayName(TextStyle.SHORT, Locale("ru"))}"
                                    return@IconButton
                                }
                                if (st.isAfter(et)) {
                                    errorMessage = "Начало должно быть раньше конца для ${setting.day.getDisplayName(TextStyle.SHORT, Locale("ru"))}"
                                    return@IconButton
                                }
                                
                                newSlots.add(
                                    ClassSlot(
                                        id = if (isEditMode) initialSlot!!.id else UUID.randomUUID().toString(),
                                        subjectTitle = subjectTitle.trim(),
                                        professor = professor.trim(),
                                        classroom = classroom.trim(),
                                        classType = classType.trim().ifBlank { "Лекция" },
                                        dayOfWeek = setting.day,
                                        startTime = st,
                                        endTime = et,
                                        weekParity = weekParity,
                                        colorHex = selectedColorHex,
                                        allowedSkips = initialSlot?.allowedSkips ?: 3,
                                        skippedCount = initialSlot?.skippedCount ?: 0
                                    )
                                )
                            }
                            onSave(newSlots)
                        }
                    ) {
                        Icon(Icons.Default.Check, contentDescription = "Сохранить", tint = BentoPrimary)
                    }
                }

                LazyColumn(
                    modifier = Modifier
                        .fillMaxSize()
                        .padding(horizontal = 20.dp),
                    contentPadding = PaddingValues(top = 16.dp, bottom = 32.dp),
                    verticalArrangement = Arrangement.spacedBy(16.dp)
                ) {
                    if (errorMessage != null) {
                        item {
                            Text(
                                text = errorMessage!!,
                                color = MaterialTheme.colorScheme.error,
                                style = MaterialTheme.typography.bodyMedium,
                                modifier = Modifier.fillMaxWidth(),
                                textAlign = TextAlign.Center
                            )
                        }
                    }

                    // Main Fields
                    item {
                        OutlinedTextField(
                            value = subjectTitle,
                            onValueChange = { subjectTitle = it },
                            label = { Text("Предмет*") },
                            modifier = Modifier.fillMaxWidth().testTag("input_subject"),
                            singleLine = true,
                            shape = RoundedCornerShape(12.dp)
                        )
                    }

                    item {
                        OutlinedTextField(
                            value = professor,
                            onValueChange = { professor = it },
                            label = { Text("Преподаватель") },
                            modifier = Modifier.fillMaxWidth().testTag("input_professor"),
                            singleLine = true,
                            shape = RoundedCornerShape(12.dp)
                        )
                    }

                    item {
                        OutlinedTextField(
                            value = classroom,
                            onValueChange = { classroom = it },
                            label = { Text("Аудитория") },
                            placeholder = { Text("Введите аудиторию либо ссылку на встречу") },
                            modifier = Modifier.fillMaxWidth().testTag("input_classroom"),
                            singleLine = true,
                            shape = RoundedCornerShape(12.dp)
                        )
                    }

                    item {
                        OutlinedTextField(
                            value = classType,
                            onValueChange = { classType = it },
                            label = { Text("Тип") },
                            modifier = Modifier.fillMaxWidth().testTag("input_type"),
                            singleLine = true,
                            shape = RoundedCornerShape(12.dp)
                        )
                    }

                    item {
                        Text(
                            text = "Цвет",
                            style = MaterialTheme.typography.labelMedium.copy(color = BentoOnSurfaceVariant)
                        )
                        Spacer(modifier = Modifier.height(8.dp))
                        LazyRow(
                            horizontalArrangement = Arrangement.spacedBy(12.dp),
                            modifier = Modifier.fillMaxWidth()
                        ) {
                            items(subjectColorPalette) { (hex, _) ->
                                val parsedColor = Color(android.graphics.Color.parseColor(hex))
                                Box(
                                    modifier = Modifier
                                        .size(44.dp)
                                        .clip(CircleShape)
                                        .background(parsedColor)
                                        .border(
                                            width = if (selectedColorHex == hex) 3.dp else 1.dp,
                                            color = if (selectedColorHex == hex) BentoOnSurface else Color.Transparent,
                                            shape = CircleShape
                                        )
                                        .clickable { selectedColorHex = hex },
                                    contentAlignment = Alignment.Center
                                ) {
                                    if (selectedColorHex == hex) {
                                        Icon(Icons.Default.Check, contentDescription = "Выбрано", tint = Color.White)
                                    }
                                }
                            }
                        }
                    }

                    item {
                        Divider(color = BentoBorderLight, modifier = Modifier.padding(vertical = 8.dp))
                        Text(
                            text = "Настройте отображение занятия*",
                            style = MaterialTheme.typography.titleMedium.copy(fontWeight = FontWeight.Bold, color = BentoOnSurface)
                        )
                        Text(
                            text = "Выберите как минимум один день либо дату",
                            style = MaterialTheme.typography.bodySmall.copy(color = BentoOnSurfaceVariant)
                        )
                    }

                    // Week Parity Options
                    item {
                        Row(
                            modifier = Modifier.fillMaxWidth(),
                            horizontalArrangement = Arrangement.spacedBy(8.dp)
                        ) {
                            WeekParity.entries.forEach { parity ->
                                val isSel = weekParity == parity
                                FilterChip(
                                    selected = isSel,
                                    onClick = { weekParity = parity },
                                    label = { Text(parity.displayName) },
                                    modifier = Modifier.weight(1f)
                                )
                            }
                        }
                    }

                    // Days selection
                    if (!isEditMode) {
                        item {
                            Text(
                                text = "Дни недели",
                                style = MaterialTheme.typography.labelMedium.copy(color = BentoOnSurfaceVariant)
                            )
                            LazyRow(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                                val days = DayOfWeek.entries
                                items(days) { day ->
                                    val isSel = daySettings.any { it.day == day }
                                    FilterChip(
                                        selected = isSel,
                                        onClick = {
                                            if (isSel) {
                                                if (daySettings.size > 1) {
                                                    daySettings = daySettings.filterNot { it.day == day }
                                                }
                                            } else {
                                                daySettings = daySettings + DayClassSetting(
                                                    day = day,
                                                    startTime = "10:00",
                                                    endTime = "11:30"
                                                )
                                            }
                                        },
                                        label = { Text(day.getDisplayName(TextStyle.SHORT, Locale("ru")).replaceFirstChar { it.uppercase() }) }
                                    )
                                }
                            }
                        }
                    }

                    // Time settings for each day
                    items(daySettings) { setting ->
                        Surface(
                            shape = RoundedCornerShape(12.dp),
                            color = BentoSurfaceVariant.copy(alpha = 0.3f),
                            border = BorderStroke(1.dp, BentoBorderLight),
                            modifier = Modifier.fillMaxWidth()
                        ) {
                            Column(modifier = Modifier.padding(12.dp)) {
                                Text(
                                    text = setting.day.getDisplayName(TextStyle.FULL, Locale("ru")).replaceFirstChar { it.uppercase() },
                                    style = MaterialTheme.typography.titleSmall.copy(fontWeight = FontWeight.Bold)
                                )
                                Spacer(modifier = Modifier.height(8.dp))
                                Row(
                                    modifier = Modifier.fillMaxWidth(),
                                    horizontalArrangement = Arrangement.spacedBy(12.dp)
                                ) {
                                    OutlinedTextField(
                                        value = setting.startTime,
                                        onValueChange = {
                                            daySettings = daySettings.map { if (it.day == setting.day) it.copy(startTime = it.startTime) else it }
                                            setting.startTime = it 
                                        },
                                        label = { Text("НАЧАЛО") },
                                        leadingIcon = { Icon(Icons.Default.AccessTime, contentDescription = null, modifier = Modifier.size(20.dp)) },
                                        modifier = Modifier.weight(1f),
                                        singleLine = true,
                                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                                        shape = RoundedCornerShape(12.dp)
                                    )
                                    OutlinedTextField(
                                        value = setting.endTime,
                                        onValueChange = {
                                            daySettings = daySettings.map { if (it.day == setting.day) it.copy(endTime = it.endTime) else it }
                                            setting.endTime = it 
                                        },
                                        label = { Text("КОНЕЦ") },
                                        modifier = Modifier.weight(1f),
                                        singleLine = true,
                                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                                        shape = RoundedCornerShape(12.dp)
                                    )
                                }
                            }
                        }
                    }

                    if (isEditMode) {
                        item {
                            Spacer(modifier = Modifier.height(16.dp))
                            FilledTonalButton(
                                onClick = { onDelete(initialSlot!!.id) },
                                colors = ButtonDefaults.filledTonalButtonColors(
                                    containerColor = MaterialTheme.colorScheme.errorContainer,
                                    contentColor = MaterialTheme.colorScheme.onErrorContainer
                                ),
                                modifier = Modifier.fillMaxWidth().testTag("btn_delete_class")
                            ) {
                                Icon(Icons.Default.Delete, contentDescription = null)
                                Spacer(modifier = Modifier.width(8.dp))
                                Text("Удалить занятие")
                            }
                        }
                    }
                }
            }
        }
    }
}
